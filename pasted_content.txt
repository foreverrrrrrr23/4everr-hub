local function isSoloIsland()
    for _, gui in pairs(player.PlayerGui:GetDescendants()) do
        if gui.Name == "MapName" and gui:IsA("TextLabel") then
            local txt = string.lower(gui.Text or "")
            if string.find(txt, "solo island") then
                return true
            end
        end
    end
    return false
end

local function checkNegativeWindDebuff()
    local windValue = 0
    pcall(function()
        local hover = player.PlayerGui:FindFirstChild("HoverPortalClick")
        if hover then
            local elem = hover:FindFirstChild("Elemental")
            if elem then
                local scroll = elem:FindFirstChild("ScrollingFrame")
                if scroll then
                    local negWind = scroll:FindFirstChild("Negative_Wind")
                    if negWind then
                        local numLabel = negWind:FindFirstChild("Number")
                        if numLabel and numLabel:IsA("TextLabel") then
                            local text = numLabel.Text
                            local extractedNum = string.match(text, "%d+")
                            if extractedNum then windValue = tonumber(extractedNum) or 0 end
                        end
                    end
                end
            end
        end
    end)
    return windValue
end

local function executePlacements()
    if _G.PlaceMode == "EquipBest" then
        pcall(function()
            local equipRemote = rs:WaitForChild("Remotes", 3):WaitForChild("Towers", 3):WaitForChild("EquipBestTowers", 3)
            if equipRemote then
                equipRemote:InvokeServer()
            end
        end)
    else
        local placeRemote = rs:WaitForChild("Remotes", 5)
        if placeRemote then placeRemote = placeRemote:WaitForChild("Towers", 5)
            if placeRemote then placeRemote = placeRemote:WaitForChild("PlaceTower", 5) end
        end
        
        local currentPreset = _G.TeamsData[_G.CurrentTeam] or {}
        if placeRemote and #currentPreset > 0 then
            for loop = 1, 2 do
                for _, args in ipairs(currentPreset) do
                    pcall(function() placeRemote:FireServer(unpack(args)) end)
                    task.wait(0.05)
                end
                task.wait(1)
            end
        end
    end
end

local function getActivePriority()
    local activePriority = {}
    for i = selectedRankIndex, #allPortals do table.insert(activePriority, allPortals[i]) end
    return activePriority
end

local function forceLoadInventory()

local function getBestPortalUUID()
    local activePriority = getActivePriority()

    for _, pName in ipairs(activePriority) do
        local candidateUUIDs = {}

        for _, obj in pairs(getgc(true)) do
            if type(obj) == "table" then
                pcall(function()
                    for key, val in pairs(obj) do
                        if type(val) == "table" then
                            local isMatch = false
                            for _, v in pairs(val) do
                                if v == pName then isMatch = true; break end
                            end

                            if isMatch then
                                local windDebuff = val.Negative_Wind or val.WindDebuff or 0
                                if type(windDebuff) == "string" then
                                    windDebuff = tonumber(string.match(windDebuff, "%d+")) or 0
                                end

                                local uuid = nil
                                if type(key) == "string" and #key >= 30 and string.find(key, "-") then uuid = key end
                                for _, v in pairs(val) do
                                    if type(v) == "string" and #v >= 30 and string.find(v, "-") then uuid = v; break end
                                end

                                if uuid then
                                    table.insert(candidateUUIDs, {uuid = uuid, wind = windDebuff})
                                end
                            end
                        end
                    end
                end)
            end
        end

        for _, portalInfo in ipairs(candidateUUIDs) do
            local uiWind = checkNegativeWindDebuff()
            local finalWind = math.max(portalInfo.wind, uiWind)

            if finalWind < maxAllowedWindDebuff then
                return portalInfo.uuid, pName, finalWind
            end
        end
    end

    return nil, nil, 0
end

-- 🔄 ลูปการทำงานหลัก (Auto Portal / Auto Place)
-- ==========================================
local function autoPortalLoop()
    task.spawn(function()
        local hasPlacedTowers = false

        while _G.AutoPortal do
            if isSoloIsland() then
                StatusLabel.Text = "In Match: Solo Island"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 170, 0)

                if _G.AutoPlace and not hasPlacedTowers then
                    if _G.PlaceMode == "EquipBest" then
                        StatusLabel.Text = "Equipping Best Towers..."
                    else
                        StatusLabel.Text = "Placing Towers (" .. _G.CurrentTeam .. ")..."
                    end
                    
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 255)
                    executePlacements()
                    
                    task.wait(1)
                    StatusLabel.Text = "Clicking Start..."
                    clickStart()
                    
                    hasPlacedTowers = true
                end
                task.wait(2)
            else
                hasPlacedTowers = false 

                StatusLabel.Text = "Loading Inventory..."
                StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
                forceLoadInventory()

                StatusLabel.Text = "Scanning Portals..."
                local uuid, portalName, windDebuff = getBestPortalUUID()

                if uuid then
                    StatusLabel.Text = "Found: " .. portalName .. " (-" .. windDebuff .. "%)"
                    StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)

                    pcall(function() activateRemote:InvokeServer(uuid) end)

                    for i = _G.StartDelay, 1, -1 do
                        StatusLabel.Text = "Waiting Alt Join: " .. i .. "s..."
                        StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
                        task.wait(1)
                    end

                    StatusLabel.Text = "Clicking Start (1)..."
                    clickStart()
                    task.wait(5)

                    StatusLabel.Text = "Clicking Start (2)..."
                    clickStart()
                    task.wait(3)
                else
                    StatusLabel.Text = "No Safe Portals! Retrying..."
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
                    task.wait(3)
                end
            end
        end

        StatusLabel.Text = "Status: Idle"
        StatusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
    end)
end

-- 🔵 ลูปการทำงาน Auto Join + Auto Place (สำหรับไอดีรอง)
-- ==========================================
local function autoJoinLoop()
    task.spawn(function()
        local hasPlacedTowersAlt = false

        while _G.AutoJoin do
            if isSoloIsland() then
                StatusLabel.Text = "In Match: Solo Island"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 170, 0)

                -- 🌟 เพิ่มระบบวางตัวสำหรับไอดีรอง (Alt Account)
                if _G.AutoPlace and not hasPlacedTowersAlt then
                    if _G.PlaceMode == "EquipBest" then
                        StatusLabel.Text = "Alt: Equipping Best Towers..."
                    else
                        StatusLabel.Text = "Alt: Placing Towers (" .. _G.CurrentTeam .. ")..."
                    end
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 255)

                    executePlacements()
                    hasPlacedTowersAlt = true
                end
                task.wait(2)
            else
                hasPlacedTowersAlt = false -- รีเซ็ตเมื่อย้อนกลับมาล็อบบี้

                pcall(function()
                    local char = player.Character or player.CharacterAdded:Wait()
                    local hrp = char:FindFirstChild("HumanoidRootPart")

                    for _, prompt in pairs(workspace:GetDescendants()) do
                        if prompt:IsA("ProximityPrompt") and _G.AutoJoin then
                            local actionText = string.lower(prompt.ActionText or "")
                            local objectText = string.lower(prompt.ObjectText or "")
                            local parentName = string.lower(prompt.Parent and prompt.Parent.Name or "")

                            if string.find(actionText, "join") or string.find(actionText, "portal") or 
                               string.find(objectText, "portal") or string.find(parentName, "portal") then

                                local parentPart = prompt.Parent:IsA("BasePart") and prompt.Parent or prompt.Parent:FindFirstChildWhichIsA("BasePart")

                                if hrp and parentPart then
                                    StatusLabel.Text = "Joining Portal..."
                                    StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 255)

                                    hrp.CFrame = parentPart.CFrame * CFrame.new(0, 0, -2)
                                    task.wait(0.1)

                                    if fireproximityprompt then
                                        fireproximityprompt(prompt)
                                    end

                                    local touch = parentPart:FindFirstChildOfClass("TouchTransmitter") or parentPart:FindFirstChild("TouchInterest")
                                    if touch then
                                        firetouchinterest(hrp, parentPart, 0)
                                        task.wait(0.1)
                                        firetouchinterest(hrp, parentPart, 1)
                                    end

                                    task.wait(3)
                                end
                            end
                        end
                    end
                end)
                task.wait(0.5)
            end
        end

        StatusLabel.Text = "Status: Idle"
        StatusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
    end)
end
