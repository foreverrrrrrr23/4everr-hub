-- ==========================================
-- ✅ FIX 1 — GC PORTAL CACHE
--
-- getgc(true) faz um full-scan da heap toda
-- vez que é chamado. No loop original isso
-- acontecia em CADA ciclo do autoPortalLoop
-- (a cada ~3s) e dentro do autoJoinLoop
-- também. Em matches longos a GC table cresce
-- — cada scan trava o scheduler por 200-600ms
-- dependendo do tamanho do heap.
--
-- Solução: escaneia UMA vez na inicialização
-- e quando o usuário clicar em Scan.
-- O cache é invalidado quando o portal
-- ativado retorna sucesso (assumindo que o
-- inventário mudou após entrar no portal).
-- ==========================================

local portalCache     = nil   -- { [uuid] = portalData }
local cacheValid      = false

local function invalidateCache()
    cacheValid = false
    portalCache = nil
end

local function copyBuff(buff)
    local entry = {}
    for k, v in pairs(buff) do entry[k] = v end
    return entry
end

local function getAllPortalsFromMemory()

    -- Retorna cache se ainda válido
    if cacheValid and portalCache then
        return portalCache
    end

    local portals = {}
    local seen    = {}

    if not getgc then
        portalCache = portals
        cacheValid  = true
        return portals
    end

    -- Scan real — só acontece quando necessário
    for _, obj in pairs(getgc(true)) do
        if type(obj) == "table" then
            pcall(function()
                if obj.IsPortal ~= true  then return end
                if not obj.PortalId      then return end
                if not obj.DisplayName   then return end

                local uuid = tostring(obj.PortalId)
                if seen[uuid] then return end
                seen[uuid] = true

                local negBuffs = {}
                local posBuffs = {}

                if type(obj.NegativeBuffs) == "table" then
                    for _, buff in pairs(obj.NegativeBuffs) do
                        if type(buff) == "table" then
                            local entry = copyBuff(buff)
                            entry._FromNegativeBuffs = true
                            if isActuallyNegativeBuff(entry) then
                                table.insert(negBuffs, entry)
                            end
                        end
                    end
                end

                if type(obj.PositiveBuffs) == "table" then
                    for _, buff in pairs(obj.PositiveBuffs) do
                        if type(buff) == "table" then
                            table.insert(posBuffs, copyBuff(buff))
                        end
                    end
                end

                table.insert(portals, {
                    uuid     = uuid,
                    name     = obj.DisplayName,
                    rank     = obj.PortalRank     or "?",
                    modifier = obj.PortalModifier or "",
                    lives    = obj.PortalLives    or 0,
                    maxLives = obj.PortalMaxLives or 0,
                    rarity   = obj.Rarity         or "",
                    negBuffs = negBuffs,
                    posBuffs = posBuffs,
                })
            end)
        end
    end

    portalCache = portals
    cacheValid  = true
    return portals
end

local function loadPresetsFromFile()
    if not readfile or not isfile then return end
    local path = folderName .. "/presets.json"
    if not isfile(path) then return end
    pcall(function()
        local decoded = HttpService:JSONDecode(readfile(path))
        if type(decoded) == "table" then
            _G.TeamsData = decoded
            _G.TeamsData.Team1 = _G.TeamsData.Team1 or {}
            _G.TeamsData.Team2 = _G.TeamsData.Team2 or {}
            _G.TeamsData.Team3 = _G.TeamsData.Team3 or {}
        end
    end)
end

local function savePresetsToFile()
    if not writefile then return end
    pcall(function()
        writefile(
            folderName .. "/presets.json",
            HttpService:JSONEncode(_G.TeamsData)
        )
    end)
end

        updateColor(newValue)
    end)

    sliderRows[elementName] = true
    updateCanvas()
end

local function getBuffValue(buff)
    if type(buff.Value)      == "number" then return buff.Value      end
    if type(buff.Amount)     == "number" then return buff.Amount     end
    if type(buff.Percentage) == "number" then return buff.Percentage end
    if type(buff.Percent)    == "number" then return buff.Percent    end
    return nil
end

local function getBuffName(buff)
    if type(buff.Name)        == "string" and buff.Name        ~= "" then return buff.Name        end
    if type(buff.DisplayName) == "string" and buff.DisplayName ~= "" then return buff.DisplayName end
    if type(buff.Element)     == "string" and buff.Element     ~= "" then return buff.Element     end
    if type(buff.Type)        == "string" and buff.Type        ~= "" then return buff.Type        end
    if type(buff.Id)          == "string" and buff.Id          ~= "" then return buff.Id          end
    return "Unknown"
end

local function hasNegativeText(value)
    if type(value) ~= "string" then return false end
    local text = string.lower(value)
    return
        string.find(text, "debuff",    1, true) ~= nil
        or string.find(text, "negative",  1, true) ~= nil
        or string.find(text, "reduction", 1, true) ~= nil
        or string.find(text, "reduced",   1, true) ~= nil
        or string.find(text, "decrease",  1, true) ~= nil
        or string.find(text, "down",      1, true) ~= nil
end

function isActuallyNegativeBuff(buff)
    if type(buff) ~= "table" then return false end

    if buff.IsDebuff  == true then return true end
    if buff.Debuff    == true then return true end
    if buff.IsNegative == true then return true end
    if buff.Negative  == true then return true end

    if type(buff.Type)     == "string" and hasNegativeText(buff.Type)     then return true end
    if type(buff.Category) == "string" and hasNegativeText(buff.Category) then return true end
    if type(buff.Class)    == "string" and hasNegativeText(buff.Class)    then return true end
    if type(buff.Name)     == "string" and hasNegativeText(buff.Name)     then return true end

    local value = getBuffValue(buff)
    if type(value) == "number" and value < 0 then return true end

    if buff._FromNegativeBuffs == true then return true end

    return false
end

-- ==========================================
-- ✅ FIX 3 — getBestPortalUUID SEM GC SCAN
--
-- Original chamava forceLoadInventory() +
-- getAllPortalsFromMemory() (getgc full scan)
-- em CADA ciclo do loop (a cada ~3-8s).
-- Agora usa o cache — GC scan só acontece
-- quando o cache está inválido.
-- Cache é invalidado após ativar um portal
-- com sucesso (inventário mudou).
-- ==========================================

local function getBestPortalUUID()
    -- Cache válido = sem GC scan
    local portals = getAllPortalsFromMemory()

    local selectedPortalName = allPortals[selectedRankIndex]
    local foundSelected = false

    for _, portal in ipairs(portals) do
        if portal.name ~= selectedPortalName then
            continue
        end

        foundSelected = true

        local passes, reason = portalPassesFilter(portal)

        if passes then
            return portal.uuid, portal.name
        else
            StatusLabel.Text      = "Skip " .. portal.name .. ": " .. reason
            StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 80)
        end
    end

    if not foundSelected then
        StatusLabel.Text       = "No " .. selectedPortalName .. " found!"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 80)
    end

    return nil, nil
end

-- ==========================================
-- ✅ FILTER PORTAL (sem mudança de lógica)
-- ==========================================

function portalPassesFilter(portal)
    for _, buff in ipairs(portal.negBuffs) do
        local elemName = getBuffName(buff)
        local rawValue = getBuffValue(buff)

        if type(rawValue) ~= "number" then
            return false, elemName .. " valor desconhecido"
        end

        local value = math.abs(rawValue)

        if not _G.DebuffLimits[elemName] then
            _G.DebuffLimits[elemName] = DEFAULT_LIMIT
        end

        if not table.find(_G.KnownNegatives, elemName) then
            table.insert(_G.KnownNegatives, elemName)
            makeSliderRow(elemName)
        end

        local limit = _G.DebuffLimits[elemName] or DEFAULT_LIMIT

        if value > limit then
            return false, elemName .. " " .. value .. "% > " .. limit .. "%"
        end
    end

    return true, ""
end

-- 🔍 SCAN INVENTORY (manual — invalida cache)
-- ==========================================

local function scanAndBuildSliders()
    StatusLabel.Text       = "Scanning inventory..."
    StatusLabel.TextColor3 = Color3.fromRGB(255, 220, 100)

    forceLoadInventory()
    task.wait(0.3)

    -- Força GC scan fresco
    invalidateCache()
    local portals = getAllPortalsFromMemory()

    local found = {}

    for _, portal in ipairs(portals) do
        for _, buff in ipairs(portal.negBuffs) do
            local name = getBuffName(buff)
            if not table.find(found, name) then
                table.insert(found, name)
            end
            if not _G.DebuffLimits[name] then
                _G.DebuffLimits[name] = DEFAULT_LIMIT
            end
            if not table.find(_G.KnownNegatives, name) then
                table.insert(_G.KnownNegatives, name)
            end
            makeSliderRow(name)
        end
    end

    if #found > 0 then
        StatusLabel.Text       = "Debuffs: " .. table.concat(found, ", ")
        StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 150)

-- 🏝️ SOLO ISLAND
-- ==========================================

local function isSoloIsland()
    for _, gui in pairs(player.PlayerGui:GetDescendants()) do
        if gui.Name == "MapName" and gui:IsA("TextLabel") then
            local text = string.lower(gui.Text or "")
            if string.find(text, "solo island", 1, true) then
                return true
            end
        end
    end
    return false
end

-- ==========================================
-- 🗼 PLACE / EQUIP TOWERS
-- ==========================================

local function executePlacements()
    if _G.PlaceMode == "EquipBest" then
        StatusLabel.Text       = "⏳ Waiting for match..."
        StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 100)

        task.wait(3)

        local Event =
            rs:WaitForChild("Remotes", 10)
            :WaitForChild("Towers", 10)
            :WaitForChild("EquipBestTowers", 10)

        for attempt = 1, 3 do
            if not _G.AutoPlace then return false end
            StatusLabel.Text = "⚙️ Equip Best (" .. attempt .. "/3)"

            local success, result = pcall(function()
                return Event:InvokeServer()
            end)

            if success then
                StatusLabel.Text       = "✅ Best Towers Equipped"
                StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 150)
                task.wait(1)
                return true
            end

            warn("[AutoPortal] EquipBest attempt " .. attempt .. " failed:", result)
            task.wait(2)
        end

        StatusLabel.Text       = "❌ Equip Best failed"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
        return false
    end

    local r = rs:WaitForChild("Remotes", 5)
    if r then r = r:WaitForChild("Towers",     5) end
    if r then r = r:WaitForChild("PlaceTower", 5) end

    local preset = _G.TeamsData[_G.CurrentTeam] or {}
    if not r or #preset == 0 then return false end

    for _ = 1, 2 do
        for _, args in ipairs(preset) do
            pcall(function() r:FireServer(unpack(args)) end)
            task.wait(0.05)
        end
        task.wait(1)
    end

    return true
end

-- ==========================================
-- 🔄 AUTO PORTAL LOOP
-- ==========================================

local function autoPortalLoop()
    task.spawn(function()
        local hasPlaced = false

        while _G.AutoPortal do
            if isSoloIsland() then
                StatusLabel.Text       = "In Match: Solo Island"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 170, 0)

                if _G.AutoPlace and not hasPlaced then
                    StatusLabel.Text       = _G.PlaceMode == "EquipBest" and "Equipping Best Towers..." or "Placing Towers..."
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 255)
                    executePlacements()
                    task.wait(1)
                    clickStart()
                    hasPlaced = true
                end

                task.wait(2)
            else
                hasPlaced = false

                StatusLabel.Text       = "Scanning Portals..."
                StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)

                local uuid, portalName = getBestPortalUUID()

                if uuid then
                    StatusLabel.Text       = "Found: " .. portalName
                    StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)

                    pcall(function()
                        activateRemote:InvokeServer(uuid)
                    end)

                    -- Portal ativado = inventário vai mudar
                    invalidateCache()

                    for i = _G.StartDelay, 1, -1 do
                        StatusLabel.Text       = "Waiting Alt Join: " .. i .. "s..."
                        StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
                        task.wait(1)
                    end

                    StatusLabel.Text = "Clicking Start (1)..."
                    clickStart()
                    task.wait(5)
                    StatusLabel.Text = "Clicking Start (2)..."
                    clickStart()
                    task.wait(3)
                else
                    StatusLabel.Text       = "No Safe Portals! Retrying..."
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
                    task.wait(3)
                end
            end
        end

        StatusLabel.Text       = "Status: Idle"
        StatusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
    end)
end

-- 🔄 AUTO JOIN LOOP (inalterado em lógica)
-- ==========================================

local function autoJoinLoop()
    task.spawn(function()
        local hasPlacedAlt = false

        while _G.AutoJoin do
            if isSoloIsland() then
                StatusLabel.Text       = "In Match: Solo Island"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 170, 0)

                if _G.AutoPlace and not hasPlacedAlt then
                    StatusLabel.Text       = _G.PlaceMode == "EquipBest" and "Alt: Equipping Best Towers..." or "Alt: Placing Towers..."
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 255)
                    executePlacements()
                    hasPlacedAlt = true
                end

                task.wait(2)
            else
                hasPlacedAlt = false

                pcall(function()
                    local char = player.Character or player.CharacterAdded:Wait()
                    local hrp  = char:FindFirstChild("HumanoidRootPart")

                    for _, prompt in pairs(workspace:GetDescendants()) do
                        if not _G.AutoJoin then break end

                        if prompt:IsA("ProximityPrompt") then
                            local action     = string.lower(prompt.ActionText or "")
                            local object     = string.lower(prompt.ObjectText  or "")
                            local parentName = string.lower(prompt.Parent and prompt.Parent.Name or "")

                            if
                                string.find(action, "join",   1, true)
                                or string.find(action, "portal", 1, true)
                                or string.find(object, "portal", 1, true)
                                or string.find(parentName, "portal", 1, true)
                            then
                                local part = prompt.Parent:IsA("BasePart")
                                    and prompt.Parent
                                    or  prompt.Parent:FindFirstChildWhichIsA("BasePart")

                                if hrp and part then
                                    StatusLabel.Text       = "Joining Portal..."
                                    StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 255)

                                    hrp.CFrame = part.CFrame * CFrame.new(0, 0, -2)
                                    task.wait(0.1)

                                    if fireproximityprompt then
                                        fireproximityprompt(prompt)
                                    end

                                    local touch =
                                        part:FindFirstChildOfClass("TouchTransmitter")
                                        or part:FindFirstChild("TouchInterest")

                                    if touch and firetouchinterest then
                                        firetouchinterest(hrp, part, 0)
                                        task.wait(0.1)
                                        firetouchinterest(hrp, part, 1)
                                    end

                                    task.wait(3)
                                end
                            end
                        end
                    end
                end)

                task.wait(0.5)
            end
        end

        StatusLabel.Text       = "Status: Idle"
        StatusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
    end)
end
