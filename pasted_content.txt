local function createToggleBtn(parent, text, initialVal, callback)
    local btn = Instance.new("TextButton")
    btn.Parent = parent; btn.Size = UDim2.new(1, 0, 0, 28); btn.Font = Enum.Font.GothamBold; btn.TextSize = 11
    local corner = Instance.new("UICorner"); corner.CornerRadius = UDim.new(0, 6); corner.Parent = btn
    local state = initialVal
    local function updateVisual()
        if state then
            btn.Text = "🟢 " .. text .. ": ON"; btn.BackgroundColor3 = Color3.fromRGB(35, 120, 60); btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            btn.Text = "🔴 " .. text .. ": OFF"; btn.BackgroundColor3 = Color3.fromRGB(38, 38, 46); btn.TextColor3 = Color3.fromRGB(170, 170, 180)
        end
    end
    updateVisual()
    btn.MouseButton1Click:Connect(function() state = not state updateVisual() callback(state) end)
    return { Button = btn, Update = function(newState) state = newState updateVisual() end }
end

local function checkRealTimeEvent()
    local t = os.date("*t")
    if t.min >= 0 and t.min < 30 then
        return true, string.format("Closes In: %02d:%02d", 29 - t.min, 59 - t.sec)
    else
        return false, string.format("Opens In: %02d:%02d", 59 - t.min, 59 - t.sec)
    end
end

local function getMapName()
    local name = ""
    pcall(function()
        if player.PlayerGui then
            for _, gui in pairs(player.PlayerGui:GetDescendants()) do
                if gui:IsA("TextLabel") and gui.Name == "MapName" then
                    name = string.lower(gui.ContentText ~= "" and gui.ContentText or gui.Text)
                    return
                end
            end
        end
    end)
    return name
end

local function placeTowers(mode, teamNum)
    if mode == "EquipBest" or mode == "Preset" then
        pcall(function() rs.Remotes.Towers.EquipBestTowers:InvokeServer() end)
    elseif mode == "TeamLoadout" then
        pcall(function() rs.Remotes.Towers.LoadTeamLoadout:InvokeServer(teamNum) end)
    end
    
    task.wait(2.5) -- รอโหลดกระเป๋า
    
    pcall(function()
        local placeRemote = rs:WaitForChild("Remotes", 3):WaitForChild("Towers", 3):WaitForChild("PlaceTower", 3)
        if not placeRemote then return end
        
        if mode == "Preset" then
            local currentPreset = _G.TeamsData[_G.CurrentTeam] or {}
            for loop = 1, 2 do
                for _, args in ipairs(currentPreset) do
                    pcall(function() placeRemote:FireServer(unpack(args)) end)
                    task.wait(0.05)
                end
                task.wait(1)
            end
        else
            for attempt = 1, 5 do
                local placed = false
                for _, v in pairs(player.PlayerGui:GetDescendants()) do
                    if v:IsA("StringValue") and string.find(string.lower(v.Name), "uuid") and #v.Value > 20 then
                        placeRemote:FireServer(v.Value, CFrame.new(0,0,0)); placed = true; break 
                    end
                end
                if placed then break end
                task.wait(1)
            end
        end
    end)
end

local function checkNegativeDarkDebuff()
    local DarkValue = 0
    pcall(function()
        local hover = player.PlayerGui:FindFirstChild("HoverPortalClick")
        if hover then
            local elem = hover:FindFirstChild("Elemental")
            if elem then
                local scroll = elem:FindFirstChild("ScrollingFrame")
                if scroll then
                    local negDark = scroll:FindFirstChild("Negative_Dark")
                    if negDark then
                        local numLabel = negDark:FindFirstChild("Number")
                        if numLabel and numLabel:IsA("TextLabel") then
                            local extractedNum = string.match(numLabel.Text, "%d+")
                            if extractedNum then DarkValue = tonumber(extractedNum) or 0 end
                        end
                    end
                end
            end
        end
    end)
    return DarkValue
end

local function getBestPortalUUID()
    local activePriority = {}
    for i = selectedRankIndex, #allPortals do table.insert(activePriority, allPortals[i]) end

    for _, pName in ipairs(activePriority) do
        local candidateUUIDs = {}
        if getgc then
            for _, obj in pairs(getgc(true)) do
                if type(obj) == "table" then
                    pcall(function()
                        for key, val in pairs(obj) do
                            if type(val) == "table" then
                                local isMatch = false
                                for _, v in pairs(val) do if v == pName then isMatch = true; break end end

                                if isMatch then
                                    local DarkDebuff = val.Negative_Dark or val.DarkDebuff or 0
                                    if type(DarkDebuff) == "string" then DarkDebuff = tonumber(string.match(DarkDebuff, "%d+")) or 0 end
                                    local uuid = nil
                                    if type(key) == "string" and #key >= 30 and string.find(key, "-") then uuid = key end
                                    for _, v in pairs(val) do if type(v) == "string" and #v >= 30 and string.find(v, "-") then uuid = v; break end end
                                    if uuid then table.insert(candidateUUIDs, {uuid = uuid, Dark = DarkDebuff}) end
                                end
                            end
                        end
                    end)
                end
            end
        end
        for _, portalInfo in ipairs(candidateUUIDs) do
            local uiDark = checkNegativeDarkDebuff()
            local finalDark = math.max(portalInfo.Dark, uiDark)
            if finalDark < maxAllowedDarkDebuff then
                return portalInfo.uuid, pName, finalDark
            end
        end
    end
    return nil, nil, 0
end

task.spawn(function()
    local portalPlaced = false

    while task.wait(1) do
        local isEventTime, timeStr = checkRealTimeEvent()
        local mapName = getMapName()
        
        local inEventMap = (string.find(mapName, "wish") or string.find(mapName, "nightmare"))
        local inRaidMap = (string.find(mapName, "raid") or string.find(mapName, "extreme raid"))
        local inPortalMap = string.find(mapName, "solo island")

        -- 🚨 PRIORITY 1: EVENT มา! ลัดคิวทุกอย่าง
        if _G.AutoEvent and isEventTime then
            _G.RaidSequenceDone = false -- ปลดล็อคให้ Raid ทำงานในรอบถัดไป
            portalPlaced = false
            
            if not inEventMap then
                -- 🌍 ถ้ายังไม่เข้า Event... เช็คก่อนว่า "เราติดอยู่ใน Raid/Portal ไหม?"
                if inRaidMap or inPortalMap then
                    local escapeTarget = _G.EscapeMaps[_G.EscapeMapIndex].value
                    local escapeName = _G.EscapeMaps[_G.EscapeMapIndex].name
                    
                    StatusLabel.Text = "🚨 Event OPEN! Escaping to " .. escapeName .. "..."
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
                    -- วาร์ปหนีออกจาก Raid ไปด่านที่กำหนด
                    pcall(function() rs.Remotes.Maps.SelectMap:FireServer(escapeTarget) end)
                    task.wait(8) -- รอโหลดแมพ 8 วิให้ชัวร์ๆ ว่าออกสำเร็จ
                end
                
                -- พออยู่หน้าด่านปกติแล้ว ค่อยสร้างห้องเข้า Event
                local diff = _G.EventDiffs[_G.EventDiffIndex]
                StatusLabel.Text = "🎃 Event | Joining " .. diff .. "..."
                pcall(function() rs.Remotes.EventDB.CreateLobby:InvokeServer(diff) end)
                task.wait(2)
                pcall(function() rs.Remotes.EventDB.Start:InvokeServer() end)
                task.wait(1)
                pcall(function() rs.Remotes.EventDB.Start:InvokeServer() end)
                
                StatusLabel.Text = "🎃 Event | Waiting 10s for map..."
                task.wait(10)
                
                if _G.AutoPlaceEvent then
                    StatusLabel.Text = "🎃 Event | Placing Towers..."
                    placeTowers(_G.PlaceModeEvent, _G.SelectedEventTeam)
                end
            else
                -- 🟢 อยู่ในด่าน Event แล้ว -> นั่งรอเกมเล่นไป
                StatusLabel.Text = "🎃 Event | Match in progress... (" .. timeStr .. ")"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 170, 0)
                
                pcall(function()
                    if player.PlayerGui then
                        for _, gui in pairs(player.PlayerGui:GetDescendants()) do
                            if (gui:IsA("TextLabel") or gui:IsA("TextButton")) and gui.Visible then
                                local text = string.lower(gui.ContentText ~= "" and gui.ContentText or gui.Text)
                                if text == "start" or text == "replay" or text == "play again" then
                                    pcall(function() rs.Remotes.EventDB.Start:InvokeServer() end)
                                    task.wait(3)
                                end
                            end
                        end
                    end
                end)
            end

        -- ⏳ OVERTIME (เวลา Event หมด แต่ตัวเรายังสู้ไม่จบในด่าน Event)
        elseif _G.AutoEvent and inEventMap then
            StatusLabel.Text = "🎃 Event | OVERTIME! Waiting to finish... " .. timeStr
            StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)

        -- ⚔️ PRIORITY 2: RAID (เมื่อไม่ได้อยู่ใน Wish Event)
        elseif _G.AutoRaid then
            if inRaidMap then
                StatusLabel.Text = "⚔️ Raid | Auto-Farming... Waiting for Event (" .. timeStr .. ")"
                StatusLabel.TextColor3 = Color3.fromRGB(200, 100, 255)
            else
                if not _G.RaidSequenceDone then
                    local diff = _G.RaidDiffs[_G.RaidDiffIndex]
                    StatusLabel.Text = "⚔️ Raid | Joining " .. diff .. " Raid..."
                    pcall(function() rs.Remotes.Maps.SelectRaid:InvokeServer(diff) end)
                    
                    StatusLabel.Text = "⚔️ Raid | Waiting 10s..."
                    task.wait(10)
                    
                    if _G.AutoPlaceRaid then
                        StatusLabel.Text = "⚔️ Raid | Placing Towers..."
                        placeTowers(_G.PlaceModeRaid, _G.SelectedRaidTeam)
                    end
                    
                    StatusLabel.Text = "⚔️ Raid | Clicking Start!"
                    task.wait(1)
                    pcall(function() rs.Remotes.Maps.StartRaid:InvokeServer() end)
                    
                    _G.RaidSequenceDone = true 
                else
                    StatusLabel.Text = "⚔️ Raid | Match Loading..."
                end
            end

        -- 🌀 PRIORITY 3: PORTAL 
        elseif _G.AutoPortal then
            if inPortalMap then
                StatusLabel.Text = "🌀 Portal | In Match"
                StatusLabel.TextColor3 = Color3.fromRGB(100, 200, 255)
                if _G.AutoPlacePortal and not portalPlaced then
                    task.wait(5)
                    placeTowers(_G.PlaceModePortal, _G.SelectedPortalTeam)
                    task.wait(1)
                    clickStartPortal()
                    portalPlaced = true
                end
            elseif inRaidMap then
                -- ถ้าไม่ได้เปิด AutoRaid แต่ดันหลงเข้าไปใน Raid
                if not _G.AutoRaid then
                    local escapeTarget = _G.EscapeMaps[_G.EscapeMapIndex].value
                    pcall(function() rs.Remotes.Maps.SelectMap:FireServer(escapeTarget) end)
                    task.wait(5)
                end
            else
                portalPlaced = false
                StatusLabel.Text = "🌀 Scanning Portals..."
                StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
                forceLoadInventory()
                local uuid, portalName, DarkDebuff = getBestPortalUUID()

                if uuid and activateRemote then
                    StatusLabel.Text = "🌀 Found: " .. portalName .. " (-" .. DarkDebuff .. "%)"
                    StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
                    
                    pcall(function()
                        if activateRemote:IsA("RemoteEvent") then activateRemote:FireServer(uuid)
                        else activateRemote:InvokeServer(uuid) end
                    end)

                    for i = _G.StartDelay, 1, -1 do
                        StatusLabel.Text = "🌀 Waiting Alt Join: " .. i .. "s"
                        task.wait(1)
                    end
                    clickStartPortal(); task.wait(5); clickStartPortal(); task.wait(3)
                else
                    StatusLabel.Text = "🌀 No safe portals. Retrying..."
                    StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
                    task.wait(3)
                end
            end
        
        else
            StatusLabel.Text = "Status: Idle (" .. timeStr .. ")"
            StatusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
        end
    end
end)

function autoJoinLoop()
    task.spawn(function()
        local hasPlacedAlt = false
        while _G.AutoJoin do
            local map = getMapName()
            if string.find(map, "solo island") then
                if _G.AutoPlacePortal and not hasPlacedAlt then
                    placeTowers(_G.PlaceModePortal, _G.SelectedPortalTeam)
                    hasPlacedAlt = true
                end
                task.wait(2)
            else
                hasPlacedAlt = false
                pcall(function()
                    local char = player.Character or player.CharacterAdded:Wait()
                    local hrp = char:FindFirstChild("HumanoidRootPart")

                    for _, prompt in pairs(workspace:GetDescendants()) do
                        if prompt:IsA("ProximityPrompt") then
                            local actionText = string.lower(prompt.ActionText or "")
                            local objectText = string.lower(prompt.ObjectText or "")
                            local parentName = string.lower(prompt.Parent and prompt.Parent.Name or "")

                            if string.find(actionText, "join") or string.find(actionText, "portal") or 
                               string.find(objectText, "portal") or string.find(parentName, "portal") then

                                local parentPart = prompt.Parent:IsA("BasePart") and prompt.Parent or prompt.Parent:FindFirstChildWhichIsA("BasePart")

                                if hrp and parentPart then
                                    hrp.CFrame = parentPart.CFrame * CFrame.new(0, 0, -2)
                                    task.wait(0.1)
                                    if fireproximityprompt then fireproximityprompt(prompt) end

                                    local touch = parentPart:FindFirstChildOfClass("TouchTransmitter") or parentPart:FindFirstChild("TouchInterest")
                                    if touch then
                                        firetouchinterest(hrp, parentPart, 0)
                                        task.wait(0.1)
                                        firetouchinterest(hrp, parentPart, 1)
                                    end
                                    task.wait(3)
                                end
                            end
                        end
                    end
                end)
                task.wait(0.5)
            end
        end
    end)
end

local function calculateSmartScore(titleText)
    local rawText = tostring(titleText or "")
    rawText = string.gsub(rawText, "<[^>]->", "") 
    local textLower = string.lower(rawText)
    local finalScore = 0
    local tierMultiplier = 1
    if string.find(textLower, " iii") or string.find(textLower, " 3") then tierMultiplier = 3
    elseif string.find(textLower, " ii") or string.find(textLower, " 2") then tierMultiplier = 2
    elseif string.find(textLower, " i") or string.find(textLower, " 1") then tierMultiplier = 1 end
    local compactText = string.gsub(textLower, "[^%w]", "")
    for cardName, info in pairs(cardDatabase) do
        local compactDbName = string.gsub(string.lower(cardName), "[^%w]", "")
        if string.find(compactText, compactDbName) then
            if info.isBuff then finalScore = info.base + ((tierMultiplier - 1) * 5)
            else
                if info.base > 0 then finalScore = info.base - ((tierMultiplier - 1) * 10)
                else finalScore = info.base end
            end
            break
        end
    end
    return finalScore
end

local function clickCardObject(obj)
    pcall(function()
        if firesignal and obj:IsA("GuiButton") then firesignal(obj.MouseButton1Click); firesignal(obj.Activated) end
        for _, conn in pairs(getconnections(obj.MouseButton1Click or obj.Activated)) do conn:Fire() end
    end)
end

    while task.wait(0.5) do

            pcall(function()
                local cards = {}
                if player.PlayerGui then
                    for _, descendant in pairs(player.PlayerGui:GetDescendants()) do
                        if descendant:IsA("TextLabel") and descendant.Name == "Title" and descendant.Visible then
                            local button = descendant:IsA("GuiButton") and descendant or descendant:FindFirstAncestorWhichIsA("GuiButton")
                            if not button then button = descendant.Parent end
                            if button then table.insert(cards, { titleLabel = descendant, button = button }) end
                        end
                    end
                end
                
                if #cards >= 3 then
                    local detectedCards = {}
                    for _, cardInfo in ipairs(cards) do
                        local rawText = cardInfo.titleLabel.ContentText ~= "" and cardInfo.titleLabel.ContentText or cardInfo.titleLabel.Text
                        table.insert(detectedCards, { title = rawText, score = calculateSmartScore(rawText), button = cardInfo.button })
                    end
                    if #detectedCards > 0 then
                        table.sort(detectedCards, function(a, b) return a.score > b.score end)
                        local bestChoice = detectedCards[1]
                        if _G.CardSelectDelay > 0 then task.wait(_G.CardSelectDelay) end
                        if _G.AutoSelectCard then clickCardObject(bestChoice.button); task.wait(2) end
                    end
                end
            end)
        end
    end
end)
