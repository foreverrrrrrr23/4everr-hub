local function calculateSmartScore(titleText)
    local rawText = tostring(titleText or ""):gsub("<[^>]->", "")
    local tier = 1

    if rawText:match("%sIII$") then
        tier = 3
    elseif rawText:match("%sII$") then
        tier = 2
    elseif rawText:match("%sI$") then
        tier = 1
    end

    local baseName = rawText:gsub("%s+[IVX]+$", "")
    local info = cardDatabase[baseName]
    if not info then return 0 end

    if info.isBuff then
        return info.base + ((tier - 1) * 5)
    end

    if info.base > 0 then
        return info.base - ((tier - 1) * 10)
    end

    return info.base
end

-- ==========================================
-- ALL REAL CARD/TIER OPTIONS

local allCardOptions = {}
local allCardOptionSet = {}
local allCardOptionByKey = {}
local romans = { [1] = "I", [2] = "II", [3] = "III" }

local function cleanCardTextBasic(text)
    text = tostring(text or "")
    text = text:gsub("<[^>]->", "")
    text = text:gsub("\194\160", " ") -- NBSP
    text = text:gsub("Ⅰ", "I")
    text = text:gsub("Ⅱ", "II")
    text = text:gsub("Ⅲ", "III")
    text = text:gsub("^%s+", "")
    text = text:gsub("%s+$", "")
    text = text:gsub("%s+", " ")
    return text
end

local function cardKey(text)
    local cleaned = string.lower(cleanCardTextBasic(text))
    -- Ignore punctuation/spacing when comparing GUI titles to config names.
    return cleaned:gsub("[^%w]", "")
end

local function rebuildCardKeyMap()
    table.clear(allCardOptionByKey)

    -- Longest names win when one key is a prefix of another (I / II / III).
    local ordered = {}
    for _, name in ipairs(allCardOptions) do
        table.insert(ordered, name)
    end

    table.sort(ordered, function(a, b)
        return #cardKey(a) > #cardKey(b)
    end)

    for _, name in ipairs(ordered) do
        allCardOptionByKey[cardKey(name)] = name
    end
end

local function addCardOption(name)
    name = cleanCardTextBasic(name)

    if type(name) ~= "string" or name == "" or allCardOptionSet[name] then
        return
    end

    allCardOptionSet[name] = true
    table.insert(allCardOptions, name)
end

local function getTierDisplayNameSafe(card, tier)
    local fallback = tostring(card.Name) .. " " .. (romans[tier] or tostring(tier))
    local getter = RaidModifierConfig and RaidModifierConfig.GetTierDisplayName

    if type(getter) ~= "function" then
        return fallback
    end

    -- Most ModuleScripts expose helpers with dot syntax. V108 used ':' which
    -- inserts self and can shift the arguments. Try dot first, then method form.
    local attempts = {
        function() return getter(card, tier) end,
        function() return getter(RaidModifierConfig, card, tier) end
    }

    local baseKey = cardKey(card.Name)

    for _, fn in ipairs(attempts) do
        local ok, result = pcall(fn)
        if ok and type(result) == "string" and result ~= "" then
            result = cleanCardTextBasic(result)
            local resultKey = cardKey(result)
            if resultKey:find(baseKey, 1, true) then
                return result
            end
        end
    end

    return fallback
end

local function buildAllCardOptions()
    table.clear(allCardOptions)
    table.clear(allCardOptionSet)

    if RaidModifierConfig and type(RaidModifierConfig.Cards) == "table" then
        for _, card in ipairs(RaidModifierConfig.Cards) do
            if type(card) == "table" and type(card.Name) == "string" then
                if type(card.Tiers) == "table" and #card.Tiers > 0 then
                    for tier = 1, #card.Tiers do
                        addCardOption(getTierDisplayNameSafe(card, tier))
                    end
                else
                    addCardOption(card.Name)
                end
            end
        end
    else
        -- Fallback built from the RaidModifierConfig you decompiled.
        local tierCounts = {
            ["Enemy HP"] = 3,
            ["Enemy Shield"] = 3,
            ["Enemy Speed"] = 3,
            ["Enemy Regen"] = 3,
            ["Boss HP"] = 3,
            ["Boss Shield"] = 3,
            ["Control Resistance"] = 3,
            ["Final Boss Trial"] = 3,
            ["Tower Damage"] = 3,
            ["Tower Attack Speed"] = 3,
            ["Tower Range"] = 3,
            ["Boss Slayer"] = 3,
            ["Domain Expansion"] = 2,
            ["Curse Overflow"] = 2,
            ["Final Nightmare"] = 2
        }

        local uniqueCards = {
            "Lucky Break",
            "Copy Blessing",
            "Cursed Copy",
            "Sealed Potential",
            "Double or Nothing",
            "One Life",
            "Heavenly Trial",
            "King's Challenge",
            "Cursed Jackpot"
        }

        for name, tierCount in pairs(tierCounts) do
            for tier = 1, tierCount do
                addCardOption(name .. " " .. romans[tier])
            end
        end

        for _, name in ipairs(uniqueCards) do
            addCardOption(name)
        end
    end

    -- Your supplied 51-entry config is the canonical default. Keep every
    -- default entry available even if the live ModuleScript helper changes.
    for _, name in ipairs(DEFAULT_CARD_PRIORITY) do
        addCardOption(name)
    end

    table.sort(allCardOptions, function(a, b)
        local baseA = a:gsub("%s+[IVX]+$", "")
        local baseB = b:gsub("%s+[IVX]+$", "")

        if baseA ~= baseB then
            return baseA < baseB
        end

        local tierValues = { I = 1, II = 2, III = 3 }
        local aTier = tierValues[a:match("([IVX]+)$")] or 0
        local bTier = tierValues[b:match("([IVX]+)$")] or 0
        return aTier < bTier
    end)

    rebuildCardKeyMap()
end

local function resolveCardOptionName(rawText)
    local cleaned = cleanCardTextBasic(rawText)
    local key = cardKey(cleaned)

    if key == "" then
        return nil
    end

    local exact = allCardOptionByKey[key]
    if exact then
        return exact
    end

    -- Some UIs add extra text around the title. Match the longest known card
    -- name contained in the label instead of rejecting the card completely.
    local bestName = nil
    local bestLen = 0

    for knownKey, knownName in pairs(allCardOptionByKey) do
        if #knownKey > bestLen and key:find(knownKey, 1, true) then
            bestName = knownName
            bestLen = #knownKey
        end
    end

    return bestName
end

local function getCardPriority(cardName)
    for i, name in ipairs(_G.CardPriority) do
        if name == cardName then
            return i
        end
    end
    return nil
end

local function sanitizePriority(list)
    local result = {}
    local seen = {}

    for _, name in ipairs(list or {}) do
        if type(name) == "string"
            and allCardOptionSet[name]
            and not seen[name] then

            seen[name] = true
            table.insert(result, name)
        end
    end

    return result
end

_G.CardPrioritySource = "default"

local function loadCardConfig()
    if readfile and isfile and isfile(cardConfigFile) then
        local loaded = false

        local ok = pcall(function()
            local decoded = HttpService:JSONDecode(readfile(cardConfigFile))
            local priority = decoded.Priority or decoded.CardPriority or decoded

            if type(priority) == "table" then
                local cleaned = sanitizePriority(priority)

                -- Empty is valid only when the saved list itself is empty.
                -- If names became invalid, keep the embedded default instead.
                if #priority == 0 or #cleaned > 0 then
                    _G.CardPriority = cleaned
                    _G.CardPrioritySource = "saved"
                    loaded = true
                end
            end
        end)

        return ok and loaded
    end

    return false
end

local function saveCardConfig()
    if not writefile then return false end

    return pcall(function()
        writefile(cardConfigFile, HttpService:JSONEncode({
            SavedAt = os.time(),
            Priority = _G.CardPriority
        }))
    end)
end

local function resetCardPriorityDefault()
    _G.CardPriority = sanitizePriority(DEFAULT_CARD_PRIORITY)
end

    _G.PlaceDelayEvent = tonumber(_G.PlaceDelayEvent) or 2
    _G.SelectedRankIndex = tonumber(_G.SelectedRankIndex) or 1
    return true

local function formatPriorityLabel(index, name)
    return string.format("#%02d  •  %s", index, name)

local function updatePriorityPreview()
    if #_G.CardPriority == 0 then
        priorityPreview:SetDesc("No manual priority. Smart Score will be used.")
        return
    end

    local lines = {}
    local limit = math.min(#_G.CardPriority, 10)

    for i = 1, limit do
        table.insert(lines, formatPriorityLabel(i, _G.CardPriority[i]))
    end

    if #_G.CardPriority > limit then
        table.insert(lines, "... +" .. tostring(#_G.CardPriority - limit) .. " more")
    end

    priorityPreview:SetDesc(table.concat(lines, "\n"))
end

local function makePositionValues(count)
    local values = {}
    for i = 1, math.max(count, 1) do
        table.insert(values, string.format("#%02d", i))
    end
    return values
end

local function moveCardToPosition(cardName, newPosition)
    local oldPosition = getCardPriority(cardName)
    if not oldPosition then return false end

    newPosition = math.clamp(tonumber(newPosition) or oldPosition, 1, #_G.CardPriority)
    if oldPosition == newPosition then return true end

    table.remove(_G.CardPriority, oldPosition)
    table.insert(_G.CardPriority, newPosition, cardName)
    return true
end

local function refreshPriorityEditor(keepCardName)
    priorityLabelToIndex = {}

    local currentValues = {}
    for i, name in ipairs(_G.CardPriority) do
        local label = formatPriorityLabel(i, name)
        priorityLabelToIndex[label] = i
        table.insert(currentValues, label)
    end

    if #currentValues == 0 then
        currentValues = {"<empty>"}
    end

    priorityDropdown:SetValues(currentValues)
    targetPositionDropdown:SetValues(makePositionValues(#_G.CardPriority))

    selectedPriorityIndex = nil
    selectedPriorityName = nil
    selectedTargetPosition = nil

    if keepCardName then
        local index = getCardPriority(keepCardName)
        if index then
            local label = formatPriorityLabel(index, keepCardName)
            priorityDropdown:SetValue(label)
            selectedPriorityIndex = index
            selectedPriorityName = keepCardName
            selectedTargetPosition = index
            targetPositionDropdown:SetValue(string.format("#%02d", index))
            selectedCardInfo:SetDesc(string.format("#%02d  •  %s", index, keepCardName))
        else
            priorityDropdown:SetValue(nil)
            targetPositionDropdown:SetValue(nil)
            selectedCardInfo:SetDesc("No card selected.")
        end
    else
        priorityDropdown:SetValue(nil)
        targetPositionDropdown:SetValue(nil)
        selectedCardInfo:SetDesc("No card selected.")
    end

    local missing = {}
    for _, name in ipairs(allCardOptions) do
        if not getCardPriority(name) then
            table.insert(missing, name)
        end
    end

    if #missing == 0 then
        missing = {"<none>"}
    end

    missingCardDropdown:SetValues(missing)
    missingCardDropdown:SetValue(nil)
    selectedMissingCard = nil

    updatePriorityPreview()
end

-- ==========================================
local function getFirstTowerUUID()
    local playerGui = player:FindFirstChild("PlayerGui")
    if not playerGui then return nil end

    for _, v in ipairs(playerGui:GetDescendants()) do
        if v:IsA("StringValue")
            and string.find(string.lower(v.Name), "uuid", 1, true)
            and type(v.Value) == "string"
            and #v.Value > 20 then
            return v.Value
        end
    end

    return nil
end

local function waitForTowerUUID(timeout)
    local started = os.clock()
    local uuid = getFirstTowerUUID()

    while not uuid and (os.clock() - started) < (timeout or 2.5) do
        task.wait(0.10)
        uuid = getFirstTowerUUID()
    end

    return uuid
end

local function placeTowers(mode, teamNum)
    local towersFolder = rs:WaitForChild("Remotes", 3):WaitForChild("Towers", 3)
    if not towersFolder then return false end

    local placeRemote = towersFolder:WaitForChild("PlaceTower", 3)
    if not placeRemote then return false end

    if mode == "Preset" then
        pcall(function()
            towersFolder.EquipBestTowers:InvokeServer()
        end)

        local currentPreset = _G.TeamsData[_G.CurrentTeam] or {}
        for _ = 1, 2 do
            for _, args in ipairs(currentPreset) do
                pcall(function()
                    placeRemote:FireServer(unpack(args))
                end)
                task.wait(0.05)
            end
            task.wait(0.35)
        end
        return true
    end

    if mode == "EquipBest" then
        -- FAST PATH: InvokeServer already yields until the server replies, so the old
        -- unconditional 1 second wait only added latency. Poll the GUI briefly instead.
        local equipped = pcall(function()
            towersFolder.EquipBestTowers:InvokeServer()
        end)

        if not equipped then
            return false
        end

        local uuid = waitForTowerUUID(2.0)

        -- One quick retry if the loadout GUI has not replicated yet.
        if not uuid then
            pcall(function()
                towersFolder.EquipBestTowers:InvokeServer()
            end)
            uuid = waitForTowerUUID(1.5)
        end

        if uuid then
            pcall(function()
                placeRemote:FireServer(uuid, CFrame.new(0, 0, 0))
            end)
            return true
        end

        return false
    end

    if mode == "TeamLoadout" then
        local loaded = pcall(function()
            towersFolder.LoadTeamLoadout:InvokeServer(teamNum or 1)
        end)

        if not loaded then
            return false
        end

        local uuid = waitForTowerUUID(2.5)
        if uuid then
            pcall(function()
                placeRemote:FireServer(uuid, CFrame.new(0, 0, 0))
            end)
            return true
        end
    end

    return false
end

-- PORTAL WIND DEBUFF
-- ==========================================
local function checkNegativeWindDebuff()
    local windValue = 0

    pcall(function()
        local hover = player.PlayerGui:FindFirstChild("HoverPortalClick")
        if hover then
            local elem = hover:FindFirstChild("Elemental")
            if elem then
                local scroll = elem:FindFirstChild("ScrollingFrame")
                if scroll then
                    local negWind = scroll:FindFirstChild("Negative_Wind")
                    if negWind then
                        local numLabel = negWind:FindFirstChild("Number")
                        if numLabel and numLabel:IsA("TextLabel") then
                            local extractedNum = string.match(numLabel.Text, "%d+")
                            if extractedNum then
                                windValue = tonumber(extractedNum) or 0
                            end
                        end
                    end
                end
            end
        end
    end)

    return windValue
end

local function getBestPortalUUID()
    local activePriority = {}
    for i = _G.SelectedRankIndex, #allPortals do
        table.insert(activePriority, allPortals[i])
    end

    for _, pName in ipairs(activePriority) do
        local candidateUUIDs = {}

        if getgc then
            for _, obj in pairs(getgc(true)) do
                if type(obj) == "table" then
                    pcall(function()
                        for key, val in pairs(obj) do
                            if type(val) == "table" then
                                local isMatch = false
                                for _, v in pairs(val) do
                                    if v == pName then
                                        isMatch = true
                                        break
                                    end
                                end

                                if isMatch then
                                    local windDebuff = val.Negative_Wind or val.WindDebuff or 0
                                    if type(windDebuff) == "string" then
                                        windDebuff = tonumber(string.match(windDebuff, "%d+")) or 0
                                    end

                                    local uuid = nil
                                    if type(key) == "string" and #key >= 30 and string.find(key, "-") then
                                        uuid = key
                                    end

                                    for _, v in pairs(val) do
                                        if type(v) == "string" and #v >= 30 and string.find(v, "-") then
                                            uuid = v
                                            break
                                        end
                                    end

                                    if uuid then
                                        table.insert(candidateUUIDs, {uuid = uuid, wind = windDebuff})
                                    end
                                end
                            end
                        end
                    end)
                end
            end
        end

        for _, portalInfo in ipairs(candidateUUIDs) do
            local uiWind = checkNegativeWindDebuff()
            local finalWind = math.max(portalInfo.wind, uiWind)
            if finalWind < maxAllowedWindDebuff then
                return portalInfo.uuid, pName, finalWind
            end
        end
    end

    return nil, nil, 0
end

local function cleanCardText(text)
    return cleanCardTextBasic(text)
end

-- Do NOT walk parent .Visible properties here.
-- In this game ModifierRaid can report Visible=false while Modifier1/2/3 are
-- still the live vote buttons. The user's debug dump confirmed this behavior.
local function readRaidCardSlot(slot)
    local playerGui = player:FindFirstChild("PlayerGui")
    local hover = playerGui and playerGui:FindFirstChild("Hover")
    local modifierRaid = hover and hover:FindFirstChild("ModifierRaid")
    local modifier = modifierRaid and modifierRaid:FindFirstChild("Modifier" .. tostring(slot))

    if not modifier or not modifier:IsA("GuiButton") then
        return nil
    end

    local frame = modifier:FindFirstChild("Frame")
    local title = frame and frame:FindFirstChild("Title")

    if not title or not title:IsA("TextLabel") then
        return nil
    end

    local rawText = ""
    pcall(function()
        rawText = (title.ContentText ~= "" and title.ContentText) or title.Text
    end)

    rawText = cleanCardText(rawText)
    if rawText == "" then
        return nil
    end

    local resolvedName = resolveCardOptionName(rawText)
    if not resolvedName then
        -- Keep the raw title as a last-resort exact candidate. This prevents a
        -- config parser issue from making the entire vote invisible to us.
        resolvedName = rawText
    end

    return {
        Name = resolvedName,
        RawName = rawText,
        TitleLabel = title,
        Button = modifier,
        Slot = slot,
        Score = calculateSmartScore(resolvedName)
    }
end

local function readAllRaidCardsDirect()
    local cards = {}

    for slot = 1, 3 do
        local card = readRaidCardSlot(slot)
        if card then
            cards[#cards + 1] = card
        end
    end

    return cards
end

local function getBestCard(cards)
    local byName = {}
    local list = {}

    for _, card in ipairs(cards or {}) do
        byName[card.Name] = card
        list[#list + 1] = card
    end

    -- Exact saved/default priority first.
    for priorityIndex, priorityName in ipairs(_G.CardPriority or {}) do
        local card = byName[priorityName]
        if card then
            card.Priority = priorityIndex
            return card
        end
    end

    -- Fallback: compare normalized raw/display names in case the game changes
    -- whitespace/rich text while keeping the same card wording.
    local function key(s)
        s = cleanCardText(tostring(s or ""))
        s = string.lower(s)
        s = s:gsub("[^%w]", "")
        return s
    end

    for priorityIndex, priorityName in ipairs(_G.CardPriority or {}) do
        local pkey = key(priorityName)
        for _, card in ipairs(list) do
            if key(card.Name) == pkey or key(card.RawName) == pkey then
                card.Priority = priorityIndex
                return card
            end
        end
    end

    -- Smart Score fallback if the priority is intentionally empty or a new
    -- card appears that is not in the current config.
    table.sort(list, function(a, b)
        return (a.Score or 0) > (b.Score or 0)
    end)

    return list[1]
end

local function describeCards(cards)
    local names = {}
    for _, card in ipairs(cards or {}) do
        names[#names + 1] = tostring(card.Slot or "?") .. ":" .. tostring(card.Name)
    end
    return table.concat(names, " | ")
end

local function getMyVoteSlot(data)
    if type(data) ~= "table" then
        return nil
    end

    for _, vote in ipairs(data.Votes or {}) do
        if tonumber(vote.UserId) == tonumber(player.UserId) then
            return tonumber(vote.OptionIndex)
        end
    end

    return nil
end

local function clickActivatedFallback(button)
    if not button or not button:IsA("GuiButton") then
        return false
    end

    local didAnything = false

    if firesignal then
        local ok = pcall(function()
            firesignal(button.Activated)
        end)
        didAnything = ok or didAnything
    end

    if getconnections then
        pcall(function()
            for _, conn in pairs(getconnections(button.Activated)) do
                if conn.Fire then
                    conn:Fire()
                    didAnything = true
                elseif conn.Function then
                    conn.Function()
                    didAnything = true
                end
            end
        end)
    end

    return didAnything
end

local function submitRaidVote(voteId, card)
    if not SubmitVoteRemote or not voteId or not card or not card.Slot then
        return false
    end

    local ok = pcall(function()
        SubmitVoteRemote:FireServer(tostring(voteId), tonumber(card.Slot))
    end)

    return ok
end

local function attemptAutoVote(voteId, trigger)
    if _G.AutoPortalHubRunId ~= RUN_ID then return end
    if not _G.AutoSelectCard then return end
    if not voteId then return end

    voteId = tostring(voteId)

    if confirmedVoteSlot[voteId] then
        return
    end

    if selectingVote[voteId] then
        return
    end

    local now = os.clock()
    local lastAttempt = lastSubmitAt[voteId] or 0

    -- If the server has not confirmed our vote yet, retry. Do not spam faster
    -- than once per second.
    if now - lastAttempt < 1.0 then
        return
    end

    selectingVote[voteId] = true

    task.spawn(function()
        local firstSeen = voteFirstSeenAt[voteId] or os.clock()
        voteFirstSeenAt[voteId] = firstSeen

        local delay = tonumber(_G.CardSelectDelay) or 0
        local remaining = delay - (os.clock() - firstSeen)

        if remaining > 0 then
            setAutoCardDebug(
                "Vote detected | " .. tostring(voteId)
                .. "\nWaiting " .. string.format("%.1f", remaining) .. "s before select..."
            )
            task.wait(remaining)
        end

        if _G.AutoPortalHubRunId ~= RUN_ID or not _G.AutoSelectCard then
            selectingVote[voteId] = nil
            return
        end

        -- Never vote an old VoteId after a new vote starts.
        if currentRaidVoteId and tostring(currentRaidVoteId) ~= voteId then
            selectingVote[voteId] = nil
            return
        end

        if confirmedVoteSlot[voteId] then
            selectingVote[voteId] = nil
            return
        end

        local cards = readAllRaidCardsDirect()

        if #cards ~= 3 then
            setAutoCardDebug(
                "VoteId found, but card reader found " .. tostring(#cards) .. "/3"
                .. "\nTrigger: " .. tostring(trigger)
            )
            selectingVote[voteId] = nil
            return
        end

        local bestChoice = getBestCard(cards)
        if not bestChoice then
            setAutoCardDebug("3 cards found, but no best choice was resolved")
            selectingVote[voteId] = nil
            return
        end

        local priorityText = bestChoice.Priority and ("#" .. tostring(bestChoice.Priority)) or "Smart Score"

        lastSubmitAt[voteId] = os.clock()
        lastSubmittedSlot[voteId] = bestChoice.Slot

        local remoteSent = submitRaidVote(voteId, bestChoice)

        setStatus("Auto Card | Sent " .. bestChoice.Name .. " | " .. priorityText)
        setAutoCardDebug(
            "SUBMITTED: " .. bestChoice.Name
            .. " | Slot " .. tostring(bestChoice.Slot)
            .. " | " .. priorityText
            .. "\nVoteId: " .. tostring(voteId)
            .. " | Remote pcall: " .. tostring(remoteSent)
            .. "\nWaiting VoteUpdate confirmation..."
        )

        print(
            "[Auto Card V111] SUBMIT",
            "VoteId:", voteId,
            "Card:", bestChoice.Name,
            "Slot:", bestChoice.Slot,
            "Priority:", priorityText,
            "RemotePcall:", remoteSent,
            "Cards:", describeCards(cards),
            "AutoRaid:", _G.AutoRaid
        )

        -- If SubmitVote is being swallowed/blocked by an executor-specific
        -- issue, also trigger the actual Activated listeners as a fallback.
        -- The user's dump showed 3 Activated connections on each modifier.
        if not remoteSent then
            local activated = clickActivatedFallback(bestChoice.Button)
            print("[Auto Card V111] Activated fallback:", activated)
        end

        selectingVote[voteId] = nil
    end)
end

if VoteUpdateRemote and VoteUpdateRemote:IsA("RemoteEvent") then
    VoteUpdateRemote.OnClientEvent:Connect(function(data)
        if _G.AutoPortalHubRunId ~= RUN_ID then return end
        if type(data) ~= "table" or not data.VoteId then return end

        local voteId = tostring(data.VoteId)
        currentRaidVoteId = voteId
        currentRaidVoteAt = os.clock()
        currentRaidVoteData = data
        voteFirstSeenAt[voteId] = voteFirstSeenAt[voteId] or os.clock()

        local mySlot = getMyVoteSlot(data)

        if mySlot then
            confirmedVoteSlot[voteId] = mySlot

            local wanted = lastSubmittedSlot[voteId]
            local status = wanted and (mySlot == wanted and "CONFIRMED" or "CONFIRMED DIFFERENT SLOT") or "ALREADY VOTED"

            setAutoCardDebug(
                status
                .. " | My slot: " .. tostring(mySlot)
                .. (wanted and (" | Wanted: " .. tostring(wanted)) or "")
                .. "\nVoteId: " .. voteId
            )

            print(
                "[Auto Card V111]", status,
                "VoteId:", voteId,
                "MySlot:", mySlot,
                "Wanted:", wanted or "none"
            )

        end

        if _G.AutoSelectCard then
            attemptAutoVote(voteId, "VoteUpdate")
        end
    end)
end

-- Polling fallback. VoteUpdate is the source of truth for VoteId/confirmation,
-- but this retry loop handles a SubmitVote that returned without being accepted.
task.spawn(function()
    while _G.AutoPortalHubRunId == RUN_ID and task.wait(0.25) do
        if not _G.AutoSelectCard then
            setAutoCardDebug(
                "OFF | Priority: " .. tostring(#(_G.CardPriority or {}))
                .. " | Source: " .. tostring(_G.CardPrioritySource or "default")
            )
            continue
        end

        if not currentRaidVoteId then
            setAutoCardDebug(
                "ON | Waiting for VoteUpdate/VoteId"
                .. " | Priority: " .. tostring(#(_G.CardPriority or {}))
            )
            continue
        end

        local voteId = tostring(currentRaidVoteId)
        local mySlot = getMyVoteSlot(currentRaidVoteData)

        if mySlot then
            confirmedVoteSlot[voteId] = mySlot
            continue
        end

        -- VoteUpdate packets in the user's log repeat throughout the voting
        -- period. Keep the current VoteId usable while packets are fresh.
        if (os.clock() - currentRaidVoteAt) <= 30 then
            attemptAutoVote(voteId, "retry-loop")
        else
            setAutoCardDebug("ON | Last VoteId is stale, waiting for next vote")
        end
    end
end)
